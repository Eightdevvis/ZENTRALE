# ZENTRALE – Python-Abhängigkeiten
# Installieren: venv/bin/pip install -r requirements.txt

# Web-Framework (Dashboard + REST API)
flask

# Tastatur-Simulation (Sensor-Simulation ohne echte Hardware, braucht sudo)
keyboard

# Spracherkennung (STT) – faster-whisper läuft auf CPU ohne GPU
faster-whisper

# Text-to-Speech (TTS) – offline
# sherpa-onnx liefert die chinesische Stimme (vits-zh-aishell3, Tutor-Modus).
# piper-tts liefert die deutsche Stimme (de_DE-kerstin-low, Haupt-Chat).
# Beide werden parallel im services/tts_service.py geladen – pro Sprache
# eine Engine. Modell-Download via services/download_tts_model.py.
sherpa-onnx
piper-tts

# Audio-Aufnahme und -Wiedergabe (für spätere Pi-Nutzung)
sounddevice
soundfile

# HTTP-Client (für Whisper/TTS Services, falls nicht über stdlib)
requests

# Recurrence-Rules (iCal RRULE) für Routinen im Kalender-System
python-dateutil

# Natives Karten-Fenster (Maps-System): scripts/map_window.py rendert die Karte
# als zweites X11-Fenster mit echter antialiased Vektorgrafik (aus der TUI per
# 'w' aufklappbar). Nur dafür nötig — TUI/Backend laufen ohne. ~15 MB Wheel.
# numpy: vektorisierte Projektion der LOD-Geometrie (10m = >500k Punkte) — ohne
# das wären feine Zoomstufen sekundenlang pro Frame. Auch nur fürs Fenster.
pygame
numpy

# Verschlüsselter Zugangsdaten-Speicher fürs Mail-System (core/mail_secrets.py).

cryptography
